defmodule Thaqqa.GlyphRegister do
  alias Thaqqa.FontRegistry, as: FontRegistry
  alias SVG.Path, as: Path
  alias Thaqqa.DataTable, as: DataTable
  alias Thaqqa.FontRegistry.Error, as: Error
  alias Thaqqa.Font, as: Font
  use Thaqqa.DataTable.Const

  def get_glyph_table(fontname) do
    case FontRegistry.get_font(fontname) do
      {:error, msg} -> {:error, msg}
      {:ok, nil} -> Error.font_not_found(fontname)
      {:ok, font} -> {:ok, font |> Font.get_table(@glyph)}
    end
  end

  def precompile_glyph() do
  end

  def precompile_glyph_all!(fontname), do: precompile_glyph!(fontname, [], true)

  def precompile_glyph!(fontname, id) when is_integer(id),
    do: precompile_glyph!(fontname, [id], false)

  def precompile_glyph!(fontname, id_list, all_flag \\ false) when is_list(id_list) do
    case precompile_glyph(fontname, id_list, all_flag) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def precompile_glyph_all(fontname), do: precompile_glyph(fontname, [], true)

  def precompile_glyph(fontname, id) when is_integer(id),
    do: precompile_glyph(fontname, [id], false)

  def precompile_glyph(fontname, id_list, all_flag \\ false) when is_list(id_list) do
    case get_glyph_table(fontname) do
      {:error, msg} ->
        {:error, msg}

      {:ok, nil} ->
        Error.table_not_found(@glyph, fontname)

      {:ok, table} ->
        table_alter =
          Enum.reduce_while(DataTable.get_data(table), table, fn %Thaqqa.Data{
                                                                   value_list: value_list
                                                                 },
                                                                 acc ->
            if Map.get(value_list, @id) in id_list or all_flag do
              path = Map.get(value_list, @path)
              path_result = SVG.Path.parse_string(path)

              case path_result do
                {:error, _msg} ->
                  {:halt, path_result}

                {:ok, reg} ->
                  value_list_ = Map.put(%{}, @precompile, reg)

                  case DataTable.edit_data(
                         acc,
                         &(&1.value_list == value_list),
                         value_list_
                       ) do
                    {:error, msg} -> {:halt, {:error, msg}}
                    {:ok, data} -> {:cont, data}
                  end
              end
            else
              {:cont, acc}
            end
          end)

        case table_alter do
          {:error, _msg} -> table_alter
          _ -> {:ok, table_alter}
        end
    end
  end

  defp validate_and_modify_precompile(value_map) do
    path = Map.get(value_map, @path)

    if path != nil do
      case SVG.Path.parse_string(path) do
        {:error, msg} -> {:error, msg}
        {:ok, val} -> {:ok, Map.put(value_map, @precompile, val)}
      end
    else
      {:ok, value_map}
    end
  end

  def add_glyph(fontname, value_map) do
    case get_glyph_table(fontname) do
      {:error, msg} ->
        {:error, msg}

      {:ok, nil} ->
        Error.table_not_found(@glyph, fontname)

      {:ok, table} ->
        case validate_and_modify_precompile(value_map) do
          {:error, msg} ->
            {:error, msg}

          {:ok, val} ->
            {:ok, table |> DataTable.add_data(val)}
        end
    end
  end

  def delete_glyph!(fontname, filter_fun, safety_num \\ 1) do
    case delete_glyph(fontname, filter_fun, safety_num) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def delete_glyph(fontname, filter_fun, safety_num \\ 1) do
    case get_glyph_table(fontname) do
      {:error, msg} -> {:error, msg}
      {:ok, nil} -> Error.table_not_found(@glyph, fontname)
      {:ok, table} -> table |> DataTable.delete_data(fontname, filter_fun, safety_num)
    end
  end

  def edit_glyph!(fontname, value_map, filter_fun, safety_num \\ 1) do
    case edit_glyph(fontname, value_map, filter_fun, safety_num) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def edit_glyph(fontname, value_map, filter_fun, safety_num \\ 1) do
    case get_glyph_table(fontname) do
      {:error, msg} ->
        {:error, msg}

      {:ok, table} ->
        case validate_and_modify_precompile(value_map) do
          {:error, msg} -> {:error, msg}
          {:ok, val} -> {:ok, table |> DataTable.edit_data(val, safety_num)}
        end
    end
  end
end

  def gpos!(string, fontname, writing_direction, gpos_type)
      when writing_direction in [@vertical, @horizontal] and gpos_type in @gpos_list do
    case gpos(string, fontname, writing_direction, gpos_type) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def gpos(string, fontname, writing_direction, gpos_type)
      when writing_direction in [@vertical, @horizontal] and gpos_type in @gpos_list do
    get_gpos(fontname, fn %Thaqqa.Data{} = data ->
      Thaqqa.Data.get_val(data, @writing_direction) in [writing_direction, @universal_direction] and
        Thaqqa.Data.get_val(data, @gpos_type) == gpos_type
    end)
    |> case do
      {:error, msg} ->
        {:error, msg}

      {:ok, val} ->
        {:ok,
         val
         |> Enum.reduce(%{x0: 0.0, y0: 0.0, x1: 0.0, y1: 0.0}, fn p, acc ->
           reg = Thaqqa.Data.get_val(p, @precompile)
           x0 = Thaqqa.Data.get_val(p, "OffsetX0")
           x1 = Thaqqa.Data.get_val(p, "OffsetX1")
           y0 = Thaqqa.Data.get_val(p, "OffsetY0")
           y1 = Thaqqa.Data.get_val(p, "OffsetY1")

           if Regex.match?(reg, string),
             do: %{x0: acc.x0 + x0, y0: acc.y0 + y0, x1: acc.x1 + x1, y1: acc.y1 + y1},
             else: acc
         end)}
    end
  end

  def sort_gpos(fontname) do
    case get_gpos_table(fontname) do
      {:error, msg} -> {:error, msg}
      {:ok, table} -> table |> Thaqqa.DataTable.sort()
    end
  end
end

    case get_parser_table() do
      {:ok, nil} -> Error.table_not_found(@parser)
      {:ok, table} -> {:ok, table |> DataTable.get_data(filter_fun)}
      {:error, msg} -> {:error, msg}
    end
  end

  def add_parser!(value_map) do
    case add_parser(value_map) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def add_parser(value_map) do
    table_ = get_parser_table()

    case table_ do
      {:error, msg} ->
        {:error, msg}

      {:ok, nil} ->
        Error.table_not_found(@parser)

      {:ok, table} ->

  def delete_parser!(filter_fun, safety_num \\ 1) do

  def edit_parser!(value_map, filter_fun, safety_num \\ 1) do
    case edit_parser(value_map, filter_fun, safety_num) do
      {:error, msg} -> raise msg

    end

    case get_parser_table() do
      {:error, msg} ->
        {:error, msg}

      {:ok, nil} ->
        Error.table_not_found(@parser)

      {:ok, table} ->
        table |> DataTable.edit_data(value_map, safety_num)
    end
  end

  def parse!(string, language) when language in [@khehyelu, @samlhta] do
    case parse(string, language) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def parse(string, language) when language in [@khehyelu, @samlhta] do
    string =
      String.split(string, "")
      |> Enum.slice(1..-2//1)
      |> Enum.map(&("R(" <> &1 <> ")"))
      |> Enum.join("")

    get_parser(fn %Thaqqa.Data{} = data ->
      Thaqqa.Data.get_val(data, @language) in [language, @universal_language]
    end)
    |> case do
      {:error, msg} ->
        {:error, msg}

      {:ok, val} ->
        {:ok,
         val
         |> Enum.reduce(string, fn p, acc ->
           reg = Thaqqa.Data.get_val(p, @precompile)
           alt = Thaqqa.Data.get_val(p, @name_alt)
           Regex.replace(reg, acc, alt)
         end)}
    end
  end

  def sort_parser() do
    case get_parser_table() do
      {:error, msg} -> {:error, msg}
      {:ok, table} -> table |> Thaqqa.DataTable.sort()
    end
  end
end

      {:ok, val} -> val
    end
  end

  def delete_unique_gsub(fontname, filter_fun, safety_num \\ 1) do
    case get_unique_gsub_table(fontname) do
      {:error, msg} -> {:error, msg}
      {:ok, nil} -> Error.table_not_found(@unique_gsub, fontname)
      {:ok, table} -> table |> DataTable.delete_data(fontname, filter_fun, safety_num)
    end
  end

  def edit_unique_gsub!(fontname, value_map, filter_fun, safety_num \\ 1) do
    case edit_unique_gsub(fontname, value_map, filter_fun, safety_num) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def edit_unique_gsub(fontname, value_map, filter_fun, safety_num \\ 1) do
    case get_unique_gsub_table(fontname) do
      {:error, msg} ->
        {:error, msg}

      {:ok, table} ->
        case validate_and_modify_precompile(value_map) do
          {:error, msg} -> {:error, msg}
          {:ok, val} -> {:ok, table |> DataTable.edit_data(val, safety_num)}
        end
    end
  end

  def unique_gsub!(string, fontname, writing_direction)
      when writing_direction in [@vertical, @horizontal] do
    case unique_gsub(string, fontname, writing_direction) do
      {:error, msg} -> raise msg
      {:ok, val} -> val
    end
  end

  def unique_gsub(string, fontname, writing_direction)
      when writing_direction in [@vertical, @horizontal] do
    get_unique_gsub(fontname, fn %Thaqqa.Data{} = data ->
      Thaqqa.Data.get_val(data, @writing_direction) in [writing_direction, @universal_direction]
    end)
    |> case do
      {:error, msg} ->
        {:error, msg}

      {:ok, val} ->
        {:ok,
         val
         |> Enum.reduce(string, fn p, acc ->
           reg = Thaqqa.Data.get_val(p, @precompile)
           alt = Thaqqa.Data.get_val(p, @name_alt)

           Regex.replace(reg, acc, alt)
         end)}
    end
  end

  def sort_unique_gsub(fontname) do
    case get_unique_gsub_table(fontname) do
      {:error, msg} -> {:error, msg}
      {:ok, table} -> table |> Thaqqa.DataTable.sort()
    end
  end
end
